find_package(SDL2 REQUIRED)
find_package(SDL2_ttf REQUIRED)
find_package(SDL2_mixer REQUIRED)

add_executable(duelo_vaqueros src/main.cpp)

set(DUELO_LIBS SDL2::SDL2)
if(TARGET SDL2::SDL2main)
    list(APPEND DUELO_LIBS SDL2::SDL2main)
endif()

if(TARGET SDL2::SDL2_ttf)
    list(APPEND DUELO_LIBS SDL2::SDL2_ttf)
elseif(TARGET SDL2_ttf::SDL2_ttf)
    list(APPEND DUELO_LIBS SDL2_ttf::SDL2_ttf)
else()
    list(APPEND DUELO_LIBS ${SDL2_TTF_LIBRARIES})
    target_include_directories(duelo_vaqueros PRIVATE ${SDL2_TTF_INCLUDE_DIRS})
endif()

if(TARGET SDL2_mixer::SDL2_mixer)
    list(APPEND DUELO_LIBS SDL2_mixer::SDL2_mixer)
elseif(TARGET SDL2::SDL2_mixer)
    list(APPEND DUELO_LIBS SDL2::SDL2_mixer)
else()
    list(APPEND DUELO_LIBS ${SDL2_MIXER_LIBRARIES})
    target_include_directories(duelo_vaqueros PRIVATE ${SDL2_MIXER_INCLUDE_DIRS})
endif()

target_link_libraries(duelo_vaqueros PRIVATE ${DUELO_LIBS})
